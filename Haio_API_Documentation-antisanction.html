نمونه کد کامل Python

import requests
import json

class HaioClient:
    def __init__(self, base_url, token):
        self.base_url = base_url
        self.headers = {
            "Authorization": f"Bearer {token}",
            "Content-Type": "application/json"
        }
    
    def get_packages(self):
        """دریافت پکیج‌های Anti Sanction"""
        url = f"{self.base_url}/v1/anti/sanction/packages"
        response = requests.get(url, headers=self.headers)
        return response.json()
    
    def create_subscription(self, title, package_id):
        """ایجاد اشتراک جدید"""
        url = f"{self.base_url}/v1/anti/sanction"
        data = {
            "title": title,
            "package_id": package_id
        }
        response = requests.post(url, json=data, headers=self.headers)
        return response.json()
    
    def get_subscriptions(self):
        """دریافت لیست اشتراک‌ها"""
        url = f"{self.base_url}/v1/anti/sanction"
        response = requests.get(url, headers=self.headers)
        return response.json()
    
    def verify_mobile(self, mobile, code):
        """تایید شماره موبایل"""
        url = f"{self.base_url}/v1/user/verifyMobile"
        data = {
            "mobile": mobile,
            "token": code
        }
        response = requests.post(url, json=data, headers=self.headers)
        return response.json()

# استفاده از کلاس
client = HaioClient("https://api.haio.cloud", "YOUR_TOKEN")

# دریافت پکیج‌ها
packages = client.get_packages()
print("پکیج‌های موجود:", packages)

# ایجاد اشتراک
new_sub = client.create_subscription("اشتراک من", 1)
print("اشتراک ایجاد شد:", new_sub)

class HaioClient {
    constructor(baseUrl, token) {
        this.baseUrl = baseUrl;
        this.headers = {
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json'
        };
    }
    
    async getPackages() {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction/packages`,
            { headers: this.headers }
        );
        return response.json();
    }
    
    async createSubscription(title, packageId) {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction`,
            {
                method: 'POST',
                headers: this.headers,
                body: JSON.stringify({ title, package_id: packageId })
            }
        );
        return response.json();
    }
    
    async getSubscriptions() {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction`,
            { headers: this.headers }
        );
        return response.json();
    }
    
    async verifyMobile(mobile, code) {
        const response = await fetch(
            `${this.baseUrl}/v1/user/verifyMobile`,
            {
                method: 'POST',
                headers: this.headers,
                body: JSON.stringify({ mobile, token: code })
            }
        );
        return response.json();
    }
}

// استفاده از کلاس
const client = new HaioClient('https://api.haio.cloud', 'YOUR_TOKEN');

// دریافت پکیج‌ها
client.getPackages().then(packages => {
    console.log('پکیج‌های موجود:', packages);
});

// ایجاد اشتراک
client.createSubscription('اشتراک من', 1).then(sub => {
    console.log('اشتراک ایجاد شد:', sub);
});

مثال ۱: احراز هویت کامل کاربر

// مثال کامل احراز هویت کاربر با Shahkar
async function authenticateUser(mobile, nationalCode) {
    try {
        // مرحله ۱: ارسال درخواست احراز هویت
        const verifyResponse = await fetch('https://api.haio.cloud/v1/user/verify/identity', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${token}`
            },
            body: JSON.stringify({ mobile, national_id: nationalCode })
        });
        
        if (!verifyResponse.ok) {
            throw new Error('خطا در ارسال درخواست');
        }
        
        // مرحله ۲: دریافت کد OTP
        const otpResponse = await fetch('https://api.haio.cloud/v1/user/getVerifyMobile', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${token}`
            },
            body: JSON.stringify({ mobile })
        });
        
        const otpData = await otpResponse.json();
        console.log('کد OTP ارسال شد');
        
        // مرحله ۳: تایید کد (کاربر باید کد را وارد کند)
        const userCode = prompt('کد ۵ رقمی را وارد کنید:');
        
        const confirmResponse = await fetch('https://api.haio.cloud/v1/user/verifyMobile', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${token}`
            },
            body: JSON.stringify({ mobile, token: userCode })
        });
        
        if (confirmResponse.ok) {
            console.log('احراز هویت موفق!');
            return true;
        } else {
            console.log('خطا در احراز هویت');
            return false;
        }
        
    } catch (error) {
        console.error('خطا:', error);
        return false;
    }
}

// استفاده
authenticateUser('09121234567', '1234567890');

مثال ۲: مدیریت اشتراک‌ها

// مثال کامل مدیریت اشتراک‌های Anti Sanction
class AntiSanctionManager {
    constructor(token) {
        this.token = token;
        this.baseUrl = 'https://api.haio.cloud';
    }
    
    async getAllPackages() {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction/packages`,
            {
                headers: {
                    'Authorization': `Bearer ${this.token}`
                }
            }
        );
        return response.json();
    }
    
    async createNewSubscription(title, packageId) {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction`,
            {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${this.token}`
                },
                body: JSON.stringify({ title, package_id: packageId })
            }
        );
        return response.json();
    }
    
    async getUserSubscriptions() {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction`,
            {
                headers: {
                    'Authorization': `Bearer ${this.token}`
                }
            }
        );
        return response.json();
    }
    
    async renewSubscription(subscriptionId) {
        const response = await fetch(
            `${this.baseUrl}/v1/anti/sanction/${subscriptionId}/renew`,
            {
                method: 'PUT',
                headers: {
                    'Authorization': `Bearer ${this.token}`
                }
            }
        );
        return response.json();
    }
}

// استفاده از کلاس
const manager = new AntiSanctionManager('YOUR_TOKEN');

// دریافت پکیج‌ها
manager.getAllPackages().then(packages => {
    console.log('پکیج‌ها:', packages);
});

// ایجاد اشتراک جدید
manager.createNewSubscription('اشتراک ویژه', 2).then(sub => {
    console.log('اشتراک ایجاد شد:', sub);
});

🛠️ عیب‌یابی

خطاهای رایج و راه‌حل‌ها

کد خطا توضیحات راه‌حل
401 توکن نامعتبر توکن خود را بررسی کنید یا توکن جدید دریافت کنید
403 عدم دسترسی سطوح دسترسی خود را بررسی کنید
404 منبع یافت نشد آدرس URL و شناسه منبع را بررسی کنید
422 خطا در داده‌ها پارامترهای ارسالی را بررسی کنید
429 درخواست زیاد صبر کنید و مجدداً تلاش کنید
500 خطای سرور با پشتیبانی تماس بگیرید

نکات مهم

  • همیشه از توکن‌های معتبر استفاده کنید
  • توکن‌ها را در کد سخت‌کد نکنید
  • از متغیرهای محیطی برای ذخیره توکن‌ها استفاده کنید
  • درخواست‌ها را با backoff مدیریت کنید
  • لاگ‌های خطا را ذخیره کنید
